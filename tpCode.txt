# Fonction qui cree une table a partir d un nombre
def iota(n):
    t = [0]*n
    for i in range(0,n):
        t[i]=i
    return t
    
# Fonction qui retourne true si le nombre appartient a la table
def contient(tab,x):
    nombreEgale = False
    for i in range(len(tab)):
        if(tab[i] == x):
            nombreEgale = True
            return nombreEgale
    return nombreEgale   
       
# Fonction qui retire un nombre d une table    
def retirer(tab,x):
    for i in range(len(tab)):
        if(tab[i] == x):
            tab.pop(i)
            return tab
           
    return tab  
        
# Fonction qui retourne une table des nombres qui sont voisins d une case
# sur une grille donne
def voisins(x,y,nX,nY):
    tailleMaxGrille = (nX * nY)-1
    tabVoisins = []
    
    voisinN = y - 1
    voisinE = x + 1
    voisinS = y + 1
    voisinO = x - 1
    
    
    if(voisinN >= 0):
        voisinN = x + voisinN * nX
        tabVoisins.append(voisinN)
        
    if(voisinE < nX):
        voisinE = voisinE + y * nX
        tabVoisins.append(voisinE)
        
    if(voisinS < nY):
        voisinS = x + voisinS * nX
        tabVoisins.append(voisinS)
        
    if(voisinO >= 0):
        voisinO = voisinO + y * nX
        tabVoisins.append(voisinO)
        
    return tabVoisins

# Fonction qui retourne une table apres avoir ajoute un element
def ajouter(tab,x):
    tab.append(x)
    return tab

# A partir des coordonnes d un case pour une grille donne, retourne le numero
# du mur choisi appartenant a la case donne
def retournerMur(x,y,nX,nY,typeMur):
    numeroMur = 0
    if(typeMur == "N"):
        numeroMur = x + y * nX
        return numeroMur
        
    if(typeMur == "E"):
        numeroMur = 1 + x + y * (nX+1)
        return numeroMur
    if(typeMur == "S"):
        numeroMur = x + (y+1) * nX
        return numeroMur
    if(typeMur == "O"):
        numeroMur = x + y * (nX+1)
        return numeroMur
    return numeroMur

#Fonction retourne les coordonnes d une case sur une grille apres
# avoir parcouru tous les nombre de la grille
def trouverCoordonnes(n,nX,nY):
    nombreTotale = nX * nY
    tabposition = [0,0]
    for i in range(0,nY):

            if(n == retournerMur(j,i,nX,nY,"N") ):
